// Модель
@Entity
class Order(
  var id: Long
  var price: Long
) {
  fun applyDiscount() {
    price = (price * 0.9).toLong()
  }
}

// Сервис приложения
class DiscountsService {
  @Transactional
  fun applyDiscount(orderId) {
    Order order = ordersRepo
                   .findByIdOrNull(orderId)
                   ?: throw NotFound()
    order.applyDiscount();
  }
}

    fun applyDiscount(orderId) {
        Order order = ordersRepo
                   .findByIdOrNull(orderId)
                   ?: throw NotFound()
        val dscntedOrder = order.applyDiscount()
        ordersRepo.save(dscntedOrder)
